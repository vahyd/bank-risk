\section{معرفی کلی}
% ===================================================================

این سامانه یک پلتفرم جامع مدیریت ریسک برای بانک‌های تجاری، بانک‌های دیجیتال و موسسات مالی است. این سیستم، اندازه‌گیری سنتی ریسک را با هوش مصنوعی، تحلیل‌های اقتصاد کلان و پیش‌بینی‌های تحلیلی ترکیب می‌کند.

\subsection{حوزه‌های تحت پوشش}
\begin{enumerate}[itemsep=4pt]
    \item ریسک اعتباری پرتفوی (Credit Portfolio Risk)
    \item ریسک تمرکز پرتفوی (Portfolio Concentration Risk)
    \item عملکرد تعدیل‌شده با ریسک (Risk-Adjusted Performance)
    \item ریسک بخشی و شرایط بازار (Sector \& Market Conditions)
    \item ریسک اقتصاد کلان (Macroeconomic Risk)
    \item ریسک نقدینگی و تأمین مالی (Liquidity \& Funding Risk)
    \item ریسک کفایت سرمایه (Capital Adequacy Risk)
    \item آزمون استرس و تحلیل سناریو (Stress Testing)
    \item سامانه هشدار زودهنگام هوش مصنوعی (AI Early Warning)
\end{enumerate}

\subsection{منابع داده}
\begin{itemize}
    \item داده‌های واقعی وام‌های \eng{LendingClub} (۲۰۰۷--۲۰۱۸) با ۱۵۱ ستون و بیش از ۲ میلیون رکورد
    \item داده‌های شبیه‌سازی‌شده بازار و اقتصاد کلان (در نبود داده واقعی)
    \item امکان بارگذاری فایل \code{CSV/Excel} توسط کاربر
\end{itemize}

\section{لایه داده (Data Layer)}
% ===================================================================

این لایه مسئول دریافت، تبدیل و آماده‌سازی داده‌ها برای موتور ریسک است.

\subsection{اولویت بارگذاری داده}

سیستم به ترتیب اولویت زیر داده‌ها را بارگذاری می‌کند:

\begin{enumerate}
    \item داده‌های موجود در \code{session\_state} (کش حافظه)
    \item داده‌های واقعی \eng{LendingClub} از پوشه \code{sample\_data/}
    \item فایل بارگذاری‌شده توسط کاربر (\code{CSV/Excel})
    \item داده‌های شبیه‌سازی‌شده (Synthetic)
\end{enumerate}

\subsection{تطبیق داده‌های \texorpdfstring{\eng{LendingClub}}{LendingClub}}

داده‌های واقعی از \eng{LendingClub} دارای ۱۵۱ ستون است که به ستون‌های استاندارد داشبورد نگاشت می‌شود:

\subsection{محاسبات مشتق}

پس از نگاشت ستون‌ها، مقادیر زیر محاسبه می‌شوند:

\begin{itemize}
    \item \textbf{احتمال نکول (PD):} بر اساس رتبه ریسک (\code{grade}) با نگاشت به مقادیر استاندارد بازل
    \item \textbf{زیان در هنگام نکول (LGD):} از روی مبلغ بازیافتی (\code{recoveries}) محاسبه می‌شود
    \item \textbf{زیان اعتباری مورد انتظار (ECL):} \lr{$PD \times EAD \times LGD$}
    \item \textbf{پرچم‌های وضعیت:} \code{is\_npl}، \code{is\_defaulted}، \code{is\_written\_off}
\end{itemize}

% ===================================================================
\section{لایه موتور ریسک (Risk Engine)}
% ===================================================================

این لایه شامل ۹ ماژول محاسباتی مستقل است که هر یک حوزه خاصی از ریسک را پوشش می‌دهد.

\subsection{ریسک اعتباری (\code{credit.py})}

شاخص‌های کلیدی:

\begin{latin}
\begin{itemize}
    \item PD (Probability of Default)
    \item EAD (Exposure at Default) 
    \item LGD (Loss Given Default)
    \item ECL = PD × EAD × LGD
    \item UL (Unexpected Loss)
    \item NPL Ratio
    \item Default Rate, Delinquency Rate, Recovery Rate
\end{itemize}
\end{latin}

\subsection{ریسک تمرکز (\code{concentration.py})}

\begin{itemize}
    \item \textbf{شاخص هرفیندال-هیرشمن (HHI):} برای وام‌گیرندگان، بخش‌ها و مناطق
    \item \textbf{نسبت تک‌وام‌گیرنده:} بزرگترین وام‌گیرنده ÷ سرمایه
    \item \textbf{سهم ۱۰ وام‌گیرنده برتر}
    \item \textbf{نسبت تنوع‌بخشی (Diversification Benefit)}
\end{itemize}

\subsection{عملکرد تعدیل‌شده با ریسک (\code{performance.py})}

\begin{latin}
\begin{itemize}
    \item RAROC = Risk-Adjusted Income ÷ Economic Capital
    \item NIM = (Interest Income − Interest Expense) ÷ Earning Assets
    \item ROA, ROE, Cost of Credit Risk
\end{itemize}
\end{latin}

\subsection{ریسک بازار (\code{market.py})}

\begin{itemize}
    \item \textbf{ارزش در معرض ریسک (VaR):} سه روش — تاریخی، پارامتریک، مونت‌کارلو
    \item \textbf{کسری مورد انتظار (Expected Shortfall / CVaR)}
    \item \textbf{حداکثر کاهش (Maximum Drawdown)}
    \item \textbf{نوسان سالانه (Annualized Volatility)}
\end{itemize}

\subsection{اقتصاد کلان (\code{macro.py})}

\begin{itemize}
    \item طبقه‌بندی چرخه اقتصادی (Expansion / Recovery / Slowdown / Recession)
    \item برآورد احتمال بحران (Crisis Probability)
    \item تحلیل حساسیت پرتفوی به عوامل کلان
\end{itemize}

\subsection{نقدینگی (\code{liquidity.py})}

\begin{itemize}
    \item \textbf{نسبت پوشش نقدینگی (LCR):} \lr{HQLA ÷ Net Cash Outflows (30-day)}
    \item \textbf{نسبت تأمین مالی پایدار (NSFR)}
    \item نردبان سررسید (Maturity Ladder)
    \item تمرکز سپرده‌ها
\end{itemize}

\subsection{کفایت سرمایه (\code{capital.py})}

\begin{itemize}
    \item \textbf{نسبت CET1:} سرمایه اصلی ÷ دارایی‌های موزون به ریسک
    \item \textbf{نسبت کفایت سرمایه کل (CAR)}
    \item \textbf{نسبت اهرمی (Leverage Ratio)}
    \item بافر حفاظتی سرمایه
\end{itemize}

\subsection{آزمون استرس (\code{stress.py})}

هشت سناریوی از پیش تعریف‌شده:

\begin{enumerate}[itemsep=2pt]
    \item \textbf{پایه (Baseline)} — بدون شوک
    \item \textbf{رکود اقتصادی} — کاهش ۴٪ تولید ناخالص داخلی
    \item \textbf{شوک نرخ بهره} — افزایش ۳۰۰ نقطه پایه
    \item \textbf{شوک تورمی} — افزایش ۵٪ تورم
    \item \textbf{بحران ارزی} — کاهش ۳۰٪ ارزش پول ملی
    \item \textbf{سقوط قیمت کالاها} — کاهش ۴۰٪ شاخص کالا
    \item \textbf{بحران مسکن} — کاهش ۳۰٪ قیمت مسکن
    \item \textbf{شوک ترکیبی شدید} — تمام شوک‌ها همزمان
\end{enumerate}

همچنین \textbf{آزمون استرس معکوس (Reverse Stress Test)} محاسبه می‌کند که چه سطحی از NPL یا کاهش GDP باعث می‌شود نسبت کفایت سرمایه به زیر حداقل نظارتی برسد.

\subsection{هوش مصنوعی (\code{ai\_models.py})}

\begin{itemize}
    \item \textbf{مدل پیش‌بینی نکول:} \eng{Gradient Boosting Classifier} با مهندسی ویژگی
    \item \textbf{تشخیص ناهنجاری:} \eng{Isolation Forest} برای شناسایی وام‌های غیرعادی
    \item \textbf{تحلیل محرک‌های ریسک:} همبستگی ویژگی‌ها با وضعیت NPL
    \item \textbf{اهمیت ویژگی (Feature Importance):} مبتنی بر مدل
    \item \textbf{تولید هشدار:} هشدارهای خودکار با سطوح بحرانی، هشدار و نظارتی
\end{itemize}

\section{لایه رابط کاربری (UI Layer)}
% ===================================================================

رابط کاربری با \eng{Streamlit} پیاده‌سازی شده و شامل ۱۰ صفحه است:

\subsection{صفحه ۱: نمای کلی مدیریتی}

نمای ۳۶۰ درجه برای مدیران ارشد:

\begin{itemize}
    \item ۶ کارت KPI اصلی (ECL، NPL، CAR، LCR، RAROC، هشدارهای AI)
    \item ماتریس حرارتی ریسک (Heatmap) — شدت ریسک در ۶ دامنه
    \item روند ۱۲ ماهه ECL و NPL
    \item هشدارهای زودهنگام هوش مصنوعی
    \item جدول ۱۰ وام‌گیرنده پرریسک
    \item جدول فضای تنفسی نظارتی (Regulatory Headroom)
\end{itemize}

\subsection{صفحه ۲: ریسک اعتباری}

\begin{itemize}
    \item کارت‌های PD، LGD، EAD، ECL، UL
    \item نرخ NPL، نرخ نکول، نرخ بازیافت، نرخ رد دین
    \item نمودار قیفی معوقات (Delinquency Funnel)
    \item توزیع PD بر اساس رتبه ریسک
    \item تحلیل Vintage — نرخ نکول بر اساس سال صدور
    \item ECL به تفکیک بخش صنعت
    \item لیست نظارتی وام‌های پرریسک (Watchlist)
\end{itemize}

\subsection{صفحه ۳: ریسک تمرکز}

\begin{itemize}
    \item شاخص HHI برای وام‌گیرندگان، بخش‌ها و مناطق
    \item نسبت بزرگترین وام‌گیرنده به سرمایه
    \item نمودار درختی (Treemap) مواجهه به تفکیک بخش
    \item نمودار میله‌ای ۲۰ وام‌گیرنده برتر
    \item نمودار دایره‌ای پراکندگی جغرافیایی
\end{itemize}

\subsection{صفحه ۴: عملکرد تعدیل‌شده با ریسک}

\begin{itemize}
    \item کارت‌های RAROC، NIM، ROA، ROE، هزینه ریسک اعتباری
    \item نمودار آبشاری (Waterfall) صورت سود و زیان
    \item جدول تفصیلی عملکرد
\end{itemize}

\subsection{صفحه ۵: شرایط بازار}

\begin{itemize}
    \item نرخ سیاستی، اسپرد ۱۰Y-2Y، اسپرد اعتباری AAA/BBB/HY
    \item نمودار منحنی بازده (Yield Curve)
    \item نرخ ارز، نوسان ارز، شاخص کالا، قیمت نفت
    \item شاخص قیمت مسکن، PMI، اعتماد مصرف‌کننده
\end{itemize}

\subsection{صفحه ۶: اقتصاد کلان}

\begin{itemize}
    \item رشد GDP، تورم، بیکاری، بدهی دولت
    \item عقربه احتمال بحران (Crisis Probability Gauge)
    \item تحلیل حساسیت پرتفوی به عوامل کلان
    \item نمودارهای روند ۸ فصل اخیر
\end{itemize}

\subsection{صفحه ۷: نقدینگی و تأمین مالی}

\begin{itemize}
    \item عقربه LCR و NSFR
    \item نردبان سررسید دارایی‌ها و بدهی‌ها
    \item نمودار دایره‌ای تفکیک سپرده‌ها
    \item جدول ۱۰ سپرده‌گذار بزرگ
\end{itemize}

\subsection{صفحه ۸: کفایت سرمایه}

\begin{itemize}
    \item کارت‌های CET1، Tier 1، CAR، اهرم
    \item نمودار آبشاری ساختار سرمایه
    \item عقربه CAR در برابر حداقل نظارتی
    \item نمودار دایره‌ای ترکیب RWA
\end{itemize}

\section{شاخص‌ها و فرمول‌های ریسک}
% ===================================================================

\subsection{ریسک اعتباری}

\begin{latin}
\begin{itemize}
    \item ECL = PD × EAD × LGD
    \item UL = ECL × multiplier (99.9\% confidence)
    \item NPL Ratio = NPL Balance ÷ Total Loans
    \item Recovery Rate = Recovered Amount ÷ Defaulted EAD
\end{itemize}
\end{latin}

\subsection{ریسک تمرکز}

\begin{latin}
\begin{itemize}
    \item HHI = \sum (exposure\_share_i)^2
    \item Diversification Benefit = 1 − \sqrt{HHI}
\end{itemize}
\end{latin}

\subsection{عملکرد تعدیل‌شده}

\begin{latin}
\begin{itemize}
    \item RAROC = (Interest Income − Expenses − Credit Losses) ÷ Economic Capital
    \item NIM = (Interest Income − Interest Expense) ÷ Earning Assets
    \item Cost of Credit = Credit Losses ÷ Average Loans
\end{itemize}
\end{latin}

\subsection{نقدینگی}

\begin{latin}
\begin{itemize}
    \item LCR = HQLA ÷ Net Cash Outflows (30-day)
    \item NSFR = Available Stable Funding ÷ Required Stable Funding
\end{itemize}
\end{latin}

\subsection{کفایت سرمایه}

\begin{latin}
\begin{itemize}
    \item CET1 Ratio = Core Capital ÷ RWA
    \item Total CAR = Total Capital ÷ RWA
    \item Leverage Ratio = Tier 1 Capital ÷ Total Exposure
\end{itemize}
\end{latin}
